cmake_minimum_required(VERSION 3.16)
project(qasr VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENABLE_CAIRO_EXPERIMENTS "Build the experimental Cairo painter" OFF)
option(ENABLE_OPENMP "Enable deterministic tile parallelism" ON)
option(ENABLE_LIBDEFLATE "Enable fast lossless PNG encoding" ON)
option(ENABLE_XML "Enable streaming vasprun.xml input with libxml2" ON)
add_library(cjson STATIC vendor/cjson/cJSON.c)
target_include_directories(cjson PUBLIC vendor/cjson)
target_compile_definitions(cjson PRIVATE CJSON_NESTING_LIMIT=32)
find_package(PNG REQUIRED)
add_library(geometry STATIC src/common.c src/structure.c src/parse.c src/config.c src/config_json.c src/atom_style.c src/automation.c src/sha256.c src/job.c src/worker.c src/elements.c src/scene.c src/render.c src/overlay.c src/png_io.c)
target_include_directories(geometry PUBLIC src)
target_compile_options(geometry PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(geometry PUBLIC cjson m PNG::PNG)
if(ENABLE_XML)
  find_package(LibXml2 2.9 REQUIRED)
  target_sources(geometry PRIVATE src/xml_parse.c)
  target_compile_definitions(geometry PUBLIC VR_HAVE_XML=1)
  target_link_libraries(geometry PUBLIC LibXml2::LibXml2)
endif()
if(ENABLE_LIBDEFLATE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBDEFLATE REQUIRED IMPORTED_TARGET libdeflate)
  target_compile_definitions(geometry PUBLIC VR_HAVE_LIBDEFLATE=1)
  target_link_libraries(geometry PUBLIC PkgConfig::LIBDEFLATE)
endif()
if(ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS C)
  target_link_libraries(geometry PUBLIC OpenMP::OpenMP_C)
endif()
if(ENABLE_CAIRO_EXPERIMENTS)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
  target_sources(geometry PRIVATE src/cairo_render.c)
  target_compile_definitions(geometry PUBLIC VR_HAVE_CAIRO=1)
  target_link_libraries(geometry PUBLIC PkgConfig::CAIRO)
endif()
add_executable(qasr src/main.c)
configure_file(src/version.h.in generated/version.h @ONLY)
foreach(schema config user job worker)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/schemas/${schema}.schema.json")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/schemas/${schema}.schema.json" schema_text HEX)
  string(REGEX REPLACE "(..)" "0x\\1," schema_bytes "${schema_text}")
  string(TOUPPER "${schema}" schema_upper)
  set(${schema_upper}_SCHEMA_BYTES "${schema_bytes}")
endforeach()
configure_file(src/schema.h.in generated/schema.h @ONLY)
target_include_directories(qasr PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_options(qasr PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(qasr PRIVATE geometry)
if(ENABLE_SANITIZERS)
  foreach(target cjson geometry qasr)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()

include(GNUInstallDirs)
install(TARGETS qasr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY presets/ DESTINATION ${CMAKE_INSTALL_DATADIR}/qasr/presets)
install(DIRECTORY schemas/ DESTINATION ${CMAKE_INSTALL_DATADIR}/qasr/schemas)
install(FILES README.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(DIRECTORY vendor/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/vendor)
install(FILES tests/fixtures/README.md tests/fixtures/PYMATGEN-LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR}/tests/fixtures)
install(FILES tests/fixtures/slabs/README.md tests/fixtures/slabs/manifest.json DESTINATION ${CMAKE_INSTALL_DOCDIR}/tests/fixtures/slabs)

set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME "qasr")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CHECKSUM SHA256)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "QASR: Quick Atomic Structure Renderer")
set(CPACK_RESOURCE_FILE_LICENSE ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE)
set(CPACK_PACKAGE_FILE_NAME "qasr-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
