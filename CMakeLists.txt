cmake_minimum_required(VERSION 3.16)
project(vasp_render VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENABLE_CAIRO_EXPERIMENTS "Build the experimental Cairo painter" OFF)
option(ENABLE_OPENMP "Enable deterministic tile parallelism" ON)
option(ENABLE_LIBDEFLATE "Enable fast lossless PNG encoding" ON)
option(ENABLE_XML "Enable streaming vasprun.xml input with libxml2" ON)
add_library(cjson STATIC vendor/cjson/cJSON.c)
target_include_directories(cjson PUBLIC vendor/cjson)
target_compile_definitions(cjson PRIVATE CJSON_NESTING_LIMIT=32)
find_package(PNG REQUIRED)
add_library(geometry STATIC src/common.c src/structure.c src/parse.c src/config.c src/elements.c src/scene.c src/render.c src/overlay.c src/png_io.c)
target_include_directories(geometry PUBLIC src)
target_compile_options(geometry PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(geometry PUBLIC cjson m PNG::PNG)
if(ENABLE_XML)
  find_package(LibXml2 2.9 REQUIRED)
  target_sources(geometry PRIVATE src/xml_parse.c)
  target_compile_definitions(geometry PUBLIC VR_HAVE_XML=1)
  target_link_libraries(geometry PUBLIC LibXml2::LibXml2)
endif()
if(ENABLE_LIBDEFLATE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBDEFLATE REQUIRED IMPORTED_TARGET libdeflate)
  target_compile_definitions(geometry PUBLIC VR_HAVE_LIBDEFLATE=1)
  target_link_libraries(geometry PUBLIC PkgConfig::LIBDEFLATE)
endif()
if(ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS C)
  target_link_libraries(geometry PUBLIC OpenMP::OpenMP_C)
endif()
if(ENABLE_CAIRO_EXPERIMENTS)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
  target_sources(geometry PRIVATE src/cairo_render.c)
  target_compile_definitions(geometry PUBLIC VR_HAVE_CAIRO=1)
  target_link_libraries(geometry PUBLIC PkgConfig::CAIRO)
endif()
add_executable(vasp-render src/main.c)
configure_file(src/version.h.in generated/version.h @ONLY)
target_include_directories(vasp-render PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_options(vasp-render PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(vasp-render PRIVATE geometry)
if(ENABLE_SANITIZERS)
  foreach(target cjson geometry vasp-render)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()

include(GNUInstallDirs)
install(TARGETS vasp-render RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY presets/ DESTINATION ${CMAKE_INSTALL_DATADIR}/vasp-render/presets)
install(FILES README.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(DIRECTORY vendor/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/vendor)
install(FILES tests/fixtures/README.md tests/fixtures/PYMATGEN-LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR}/tests/fixtures)
install(FILES tests/fixtures/slabs/README.md tests/fixtures/slabs/manifest.json DESTINATION ${CMAKE_INSTALL_DOCDIR}/tests/fixtures/slabs)

set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME "vasp-render")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CHECKSUM SHA256)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Headless VASP geometry-to-PNG renderer")
set(CPACK_RESOURCE_FILE_LICENSE ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE)
set(CPACK_PACKAGE_FILE_NAME "vasp-render-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(png-encode-bench tools/png_bench.c)
  target_link_libraries(png-encode-bench PRIVATE geometry)
  target_compile_options(png-encode-bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(ENABLE_SANITIZERS)
    target_compile_options(png-encode-bench PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  add_executable(png-budget-gate tests/test_png_limits.c)
  target_link_libraries(png-budget-gate PRIVATE geometry)
  target_compile_options(png-budget-gate PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(ENABLE_SANITIZERS)
    target_compile_options(png-budget-gate PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  add_test(NAME encoders COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_encoders.py $<TARGET_FILE:vasp-render> $<TARGET_FILE:png-encode-bench>)
  add_test(NAME png_budget COMMAND png-budget-gate)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:vasp-render> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  add_test(NAME scene_render COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_scene_render.py $<TARGET_FILE:vasp-render>)
  add_test(NAME benchmark_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark_tools.py)
  add_test(NAME overlays COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_overlays.py $<TARGET_FILE:vasp-render>)
  add_test(NAME xml_input COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_xml.py $<TARGET_FILE:vasp-render>)
  add_test(NAME periodic_context COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_periodic_context.py $<TARGET_FILE:vasp-render>)
  if(ENABLE_CAIRO_EXPERIMENTS)
    add_test(NAME painter_experiments COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_experiments.py $<TARGET_FILE:vasp-render>)
    add_executable(painter-depth-gate tests/test_painter_depth.c)
    target_link_libraries(painter-depth-gate PRIVATE geometry)
    target_compile_options(painter-depth-gate PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(ENABLE_SANITIZERS)
      target_compile_options(painter-depth-gate PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME painter_depth_rejection COMMAND painter-depth-gate)
  endif()
endif()
